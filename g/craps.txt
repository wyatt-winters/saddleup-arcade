function oddsR(n){return n===4||n===10?[2,1]:n===5||n===9?[3,2]:[6,5];}
function layR(n){return n===4||n===10?[1,2]:n===5||n===9?[2,3]:[5,6];}
function placeR(n){return n===4||n===10?[9,5]:n===5||n===9?[7,5]:[7,6];}
function pays(a,r){return Math.floor(a*r[0]/r[1]+1e-9);}
function oddsMult(n){if(opt.odds==='10')return 10;if(opt.odds==='100')return 100;return n===4||n===10?3:n===5||n===9?4:5;}
function maxOdds(flat,n){return flat*oddsMult(n);}
function maxLay(flat,n){var r=layR(n);return Math.floor(flat*oddsMult(n)*r[1]/r[0]+1e-9);}

function canAdd(pl,k){
  var b=pl.bets;
  if(k==='pass'||k==='dp')return point===0;
  if(k==='passO')return point>0&&b.pass>0;
  if(k==='dpO')return point>0&&b.dp>0;

  if(/^Do\d/.test(k))return b['D'+nOf(k)]>0;
  if(/^[CD]\d/.test(k))return false;

function addCap(pl,k,amt){
  var b=pl.bets,cur=b[k]||0,n=nOf(k);
  if(k==='passO')return Math.max(0,Math.min(amt,maxOdds(b.pass,point)-cur));
  if(k==='dpO')return Math.max(0,Math.min(amt,maxLay(b.dp,point)-cur));
  if(/^Co\d/.test(k))return Math.max(0,Math.min(amt,maxOdds(b['C'+n],n)-cur));
  if(/^Do\d/.test(k))return Math.max(0,Math.min(amt,maxLay(b['D'+n],n)-cur));
  if(k==='P6'||k==='P8'){var tgt=Math.ceil((cur+amt)/6)*6;return tgt-cur;}
  return amt;
}

/* settle one roll for every player at the table */
function resolve(d1,d2){
  var s=d1+d2,hard=d1===d2,P=point,co=P===0,out=[];
  players.forEach(function(pl){
    var b=pl.bets,res=[],moves=[];
    function lose(k){if(!b[k])return;res.push({k:k,type:'lose',amt:b[k],net:-b[k]});b[k]=0;}
    function stay(k,w){if(!b[k])return;res.push({k:k,type:'win',amt:b[k],pay:w,net:w,down:false});pl.bank+=w;}
    function down(k,w){if(!b[k])return;res.push({k:k,type:'win',amt:b[k],pay:w,net:w,down:true});pl.bank+=w+b[k];b[k]=0;}
    function back(k,why){if(!b[k])return;res.push({k:k,type:'ret',amt:b[k],net:0,why:why});pl.bank+=b[k];b[k]=0;}
    function push(k){if(!b[k])return;res.push({k:k,type:'push',amt:b[k],net:0});}
    /* line bets */
    if(b.pass){if(co){if(s===7||s===11)stay('pass',b.pass);else if(s===2||s===3||s===12)lose('pass');}
      else{if(s===P)stay('pass',b.pass);else if(s===7)lose('pass');}}
    if(b.passO&&!co){if(s===P)down('passO',pays(b.passO,oddsR(P)));else if(s===7)lose('passO');}
    if(b.dp){if(co){if(s===2||s===3)stay('dp',b.dp);else if(s===12)push('dp');else if(s===7||s===11)lose('dp');}
      else{if(s===7)stay('dp',b.dp);else if(s===P)lose('dp');}}
    if(b.dpO&&!co){if(s===7)down('dpO',pays(b.dpO,layR(P)));else if(s===P)lose('dpO');}
    /* come and don't come points (settled before the come box moves) */
    PT.forEach(function(n){
      var C='C'+n,Co='Co'+n,D='D'+n,Do='Do'+n;
      if(b[C]){if(s===n){down(C,b[C]);if(b[Co]){if(co)back(Co,'off');else down(Co,pays(b[Co],oddsR(n)));}}
        else if(s===7){lose(C);if(b[Co]){if(co)back(Co,'off');else lose(Co);}}}
      else if(b[Co])back(Co,'');
      if(b[D]){if(s===7){down(D,b[D]);if(b[Do])down(Do,pays(b[Do],layR(n)));}else if(s===n){lose(D);lose(Do);}}
      else if(b[Do])back(Do,'');
    });
    if(b.come){if(s===7)down('come',b.come);else if(s===11)stay('come',b.come);else if(s===2||s===3||s===12)lose('come');
      else{moves.push({from:'come',to:'C'+s,amt:b.come});b['C'+s]=(b['C'+s]||0)+b.come;b.come=0;}}
    if(b.dc){if(s===2||s===3)stay('dc',b.dc);else if(s===12)push('dc');else if(s===7||s===11)lose('dc');
      else{moves.push({from:'dc',to:'D'+s,amt:b.dc});b['D'+s]=(b['D'+s]||0)+b.dc;b.dc=0;}}
    /* place, buy, lay */
    var work=!co||opt.work;
    PT.forEach(function(n){
      var Pk='P'+n,Bk='B'+n,Lk='L'+n;
      if(b[Pk]&&work){if(s===n)stay(Pk,pays(b[Pk],placeR(n)));else if(s===7)lose(Pk);}
      if(b[Bk]&&work){if(s===n){var w=pays(b[Bk],oddsR(n)),vig=Math.max(1,Math.floor(b[Bk]*0.05));stay(Bk,Math.max(0,w-vig));}else if(s===7)lose(Bk);}
      if(b[Lk]){if(s===7){var w2=pays(b[Lk],layR(n)),vig2=Math.max(1,Math.floor(w2*0.05));stay(Lk,Math.max(0,w2-vig2));}else if(s===n)lose(Lk);}
    });
    /* one-roll and multi-roll center bets */
    if(b.field){if(s===2)stay('field',b.field*2);else if(s===12)stay('field',b.field*opt.f12);else if(s===3||s===4||s===9||s===10||s===11)stay('field',b.field);else lose('field');}
    if(b.big6){if(s===6)stay('big6',b.big6);else if(s===7)lose('big6');}
    if(b.big8){if(s===8)stay('big8',b.big8);else if(s===7)lose('big8');}
    [4,6,8,10].forEach(function(n){var k='h'+n;if(!b[k]||!work)return;
      if(s===n&&hard)stay(k,b[k]*(n===4||n===10?7:9));else if(s===7||s===n)lose(k);});
    if(b.any7){if(s===7)stay('any7',b.any7*4);else lose('any7');}
    if(b.any){if(s===2||s===3||s===12)stay('any',b.any*7);else lose('any');}
    if(b.y2){if(s===2)stay('y2',b.y2*30);else lose('y2');}
    if(b.y12){if(s===12)stay('y12',b.y12*30);else lose('y12');}
    if(b.y3){if(s===3)stay('y3',b.y3*15);else lose('y3');}
    if(b.y11){if(s===11)stay('y11',b.y11*15);else lose('y11');}
    if(b.horn){var u=b.horn/4;if(s===2||s===12)stay('horn',Math.floor(u*27+1e-9));else if(s===3||s===11)stay('horn',Math.floor(u*12+1e-9));else lose('horn');}
    if(b.ce){if(s===2||s===3||s===12)stay('ce',b.ce*3);else if(s===11)stay('ce',b.ce*7);else lose('ce');}
    for(var k in b)if(!b[k])delete b[k];
    var net=0;res.forEach(function(r){net+=r.net;});
    out.push({pl:pl,res:res,moves:moves,net:net});
  });
  var made=false,out7=false,est=0;
  if(co){if(PT.indexOf(s)>=0){point=s;est=s;}}
  else if(s===P){point=0;made=true;}
  else if(s===7){point=0;out7=true;}
  return {list:out,s:s,d1:d1,d2:d2,hard:hard,P:P,co:co,made:made,out7:out7,est:est};
}

  function axI(v){for(var i=0;i<6;i++)if(AX[i][0]===v[0]&&AX[i][1]===v[1]&&AX[i][2]===v[2])return i;return -1;}
  function rotA(i,r){var v=AX[i],x=v[0],y=v[1],z=v[2];if(r==='x')return axI([x,-z,y]);if(r==='y')return axI([z,y,-x]);return axI([-y,x,z]);}

    ['x','y','z'].forEach(function(r){var m=[0,0,0,0,0,0];for(var i=0;i<6;i++)m[rotA(i,r)]=n[i];q.push(m);});}return out;})();
  function qm(a,b){return [a[0]*b[0]-a[1]*b[1]-a[2]*b[2]-a[3]*b[3],a[0]*b[1]+a[1]*b[0]+a[2]*b[3]-a[3]*b[2],a[0]*b[2]-a[1]*b[3]+a[2]*b[0]+a[3]*b[1],a[0]*b[3]+a[1]*b[2]-a[2]*b[1]+a[3]*b[0]];}
  function qa(ax,ang){var l=Math.hypot(ax[0],ax[1],ax[2])||1,s=Math.sin(ang/2);return [Math.cos(ang/2),ax[0]/l*s,ax[1]/l*s,ax[2]/l*s];}

  function qrot(q,v){var m=qmat(q);return [m[0]*v[0]+m[1]*v[1]+m[2]*v[2],m[3]*v[0]+m[4]*v[1]+m[5]*v[2],m[6]*v[0]+m[7]*v[1]+m[8]*v[2]];}

function bput(pl,k,amt){
  amt=Math.floor(amt);if(amt<=0||!canAdd(pl,k))return 0;amt=addCap(pl,k,amt);if(amt<=0||amt>pl.bank)return 0;
  pl.bank-=amt;pl.bets[k]=(pl.bets[k]||0)+amt;return amt;
}
function oddsAmt(n,base,lay){
  if(lay){var r=layR(n);return Math.ceil(base*r[1]/r[0]/r[1])*r[1];}
  return n===6||n===8?Math.ceil(base/5)*5:n===5||n===9?Math.ceil(base/2)*2:base;
}
function botDecide(pl){
  var b=pl.bets,u=pl.unit,P=point,placed=[],k;
  function put(key,amt){var a=bput(pl,key,amt);if(a)placed.push([key,a]);}
  var cnt=function(re){return Object.keys(b).filter(function(x){return re.test(x)&&b[x];}).length;};
  var poor=pl.bank<u*3;
  switch(pl.kind){
    case 'grinder':
      if(!P){if(!b.pass)put('pass',u);}
      else{if(b.pass&&!b.passO)put('passO',oddsAmt(P,u*(poor?1:2)));
        if(!b.come&&cnt(/^C\d/)<2&&R()<0.65&&!poor)put('come',u);
        PT.forEach(function(n){if(b['C'+n]&&!b['Co'+n]&&R()<0.85)put('Co'+n,oddsAmt(n,u*2));});}
      break;
    case 'dark':
      if(!P){if(!b.dp)put('dp',u);}
      else{if(b.dp&&!b.dpO)put('dpO',oddsAmt(P,u*2,true));
        if(!b.dc&&cnt(/^D\d/)<1&&R()<0.45&&!poor)put('dc',u);
        PT.forEach(function(n){if(b['D'+n]&&!b['Do'+n])put('Do'+n,oddsAmt(n,u*2,true));});}
      break;
    case 'presser':
      if(!P){if(!b.pass&&R()<0.7)put('pass',Math.max(5,Math.round(u/6)*5));}
      else{if(!b.P6)put('P6',u);if(!b.P8)put('P8',u);
        if(!b.P5&&R()<0.25)put('P5',Math.round(u/6)*5||5);if(!b.P9&&R()<0.25)put('P9',Math.round(u/6)*5||5);
        ['P6','P8','P5','P9'].forEach(function(x){if(pl.lastWin[x]&&b[x]&&R()<0.6)put(x,x==='P6'||x==='P8'?6*Math.max(1,Math.round(b[x]/12)):5*Math.max(1,Math.round(b[x]/10)));});}
      break;
    case 'degen':
      if(R()<0.55&&!b.field)put('field',u);
      if(!P){if(R()<0.35&&!b.horn)put('horn',4*Math.max(1,Math.round(u/4)));else if(R()<0.3&&!b.y11)put('y11',u);else if(R()<0.25&&!b.ce)put('ce',u);
        if(!b.pass&&R()<0.5)put('pass',u);}
      else{if(!b.h6&&R()<0.7)put('h6',u);if(!b.h8&&R()<0.7)put('h8',u);if(!b.h4&&R()<0.25)put('h4',u);if(!b.h10&&R()<0.25)put('h10',u);
        if(R()<0.12&&!b.y12)put('y12',u);if(R()<0.08&&!b.any7)put('any7',u);}
      break;
    case 'whale':
      if(!P){if(!b.pass)put('pass',u);if(R()<0.3&&!b.any)put('any',Math.max(5,Math.round(u/10/5)*5));}
      else{if(b.pass&&!b.passO)put('passO',oddsAmt(P,Math.min(maxOdds(b.pass,P),u*3)));
        if(!b.P6&&P!==6)put('P6',Math.round(u*1.2/6)*6);if(!b.P8&&P!==8)put('P8',Math.round(u*1.2/6)*6);
        if(!b.h6&&R()<0.5)put('h6',Math.round(u/4/5)*5||5);if(!b.h8&&R()<0.5)put('h8',Math.round(u/4/5)*5||5);
        if(!b.come&&cnt(/^C\d/)<1&&R()<0.4)put('come',u);
        PT.forEach(function(n){if(b['C'+n]&&!b['Co'+n])put('Co'+n,oddsAmt(n,u*2));});}
      break;
  }
  pl.lastWin={};
  return placed;
}

  var fs=freeSpots();

function composeMsg(call,mine,r0){
  var head='<b'+(call.k==='bad'?' class="l"':'')+'>'+esc(call.t)+'</b>';
  if(!mine||(!mine.res.length&&!mine.moves.length)){if(r0&&r0.est&&me.bets.pass)return head+' <em>Your Pass Line rides on the '+r0.est+'. Tap it again to take free odds.</em>';if(r0&&r0.est&&me.bets.dp)return head+' <em>Your Don\u2019t Pass is working. Tap it again to lay odds.</em>';
    return head+(onTable(me)?' <em>No action on your bets.</em>':'');}
  var parts=mine.res.map(function(it){
    if(it.type==='push')return bname(it.k)+' push';
    if(it.type==='ret')return bname(it.k)+' returned';
    return bname(it.k)+' '+smoney(it.net);});
  mine.moves.forEach(function(mv){parts.push(bname(mv.from)+' moves to '+nOf(mv.to));});
  var n=mine.net,h=n>0?' You win <b>'+money(n)+'</b> net':n<0?' You lose <b class="l">'+money(-n)+'</b> net':' You break even';
  if(!mine.res.some(function(it){return it.net!==0;}))h='';
  return head+h+' <em>('+esc(parts.join(' \u00b7 '))+')</em>';
}

}
