/* build a solvable arrangement: remove random free pairs from the full (or remaining) shape, then give each removed pair matching faces */
function arrange(list,faces){
  /* faces: array of face ids (even counts per group). returns true when assigned */
  var pairs=[],byG={};faces.forEach(function(f){var g=grp(f);(byG[g]=byG[g]||[]).push(f);});
  Object.keys(byG).forEach(function(g){var a=shuffleArr(byG[g]);for(var i=0;i+1<a.length;i+=2)pairs.push([a[i],a[i+1]]);});
  for(var tries=0;tries<400;tries++){
    list.forEach(function(t){t.gone=false;});
    var order=[],left=list.length,ok=true;
    while(left>0){
      var fr=list.filter(function(t){return !t.gone&&isFree(t);});
      if(fr.length<2){ok=false;break;}
      /* lean towards high and outer tiles first, so the shape opens up like it does in play */
      var a=fr[ri(fr.length)],b;do{b=fr[ri(fr.length)];}while(b===a);
      a.gone=true;b.gone=true;order.push([a,b]);left-=2;
    }
    list.forEach(function(t){t.gone=false;});
    if(!ok)continue;
    shuffleArr(pairs);
    order.forEach(function(p,i){var fp=pairs[i];if(R()<.5)fp=[fp[1],fp[0]];p[0].f=fp[0];p[1].f=fp[1];});
    plan=order.map(function(p){return [p[0].id,p[1].id];});
    return true;
  }
  return false;
}
