/* =====================================================================
   Rules engine. A position is an Int8Array(52): side s owns indexes s*26..s*26+25,
   counted from that side's own point of view: 1..24 = points (1 = deepest home point),
   25 = bar, 0 = borne off. Opponent checkers on my point i sit on their point 25-i.
   ===================================================================== */
function newPos(){var p=new Int8Array(52);for(var s=0;s<2;s++){var b=s*26;p[b+24]=2;p[b+13]=5;p[b+8]=3;p[b+6]=5;}return p;}

function allHome(p,s){var b=s*26;if(p[b+25])return false;for(var i=7;i<=24;i++)if(p[b+i])return false;return true;}
function singles(p,s,d,out){
  var b=s*26,o=(1-s)*26;
  if(p[b+25]>0){var t0=25-d;if(p[o+25-t0]<2)out.push({f:25,t:t0,d:d});return out;}
  var home=allHome(p,s),hi=0,f;
  if(home)for(f=6;f>=1;f--)if(p[b+f]){hi=f;break;}
  for(f=24;f>=1;f--){if(!p[b+f])continue;var t=f-d;
    if(t>=1){if(p[o+25-t]<2)out.push({f:f,t:t,d:d});}
    else if(home&&(t===0||f===hi))out.push({f:f,t:0,d:d});}
  return out;
}

var PTV=[0,.06,.12,.2,.27,.34,.3,.26,.16,.1,.08,.06,.05,.04,.04,.04,.04,.06,.12,.16,.24,.2,.12,.08,.06];

function shotRisk(p,s){
  var b=s*26,o=(1-s)*26,blots=[],srcs=[],i;
  for(i=1;i<=24;i++)if(p[b+i]===1)blots.push(i);
  if(!blots.length)return 0;
  var onBar=p[o+25]>0;
  for(i=1;i<=25;i++)if(p[o+i]&&(!onBar||i===25))srcs.push(25-i); /* in my numbering; 0 = their bar */
  var oh=homePts(p,1-s),tot=0;
  function blocked(x){return x>=1&&x<=24&&p[b+x]>=2;}
  ROLLS.forEach(function(r){
    var d1=r.d[0],d2=r.d[1],dbl=d1===d2,worst=0;
    blots.forEach(function(bi){
      var hit=false;
      for(var k=0;k<srcs.length&&!hit;k++){var x=srcs[k],dist=bi-x;if(dist<=0)continue;
        if(dist===d1||dist===d2)hit=true;
        else if(!dbl&&dist===d1+d2&&(!blocked(x+d1)||!blocked(x+d2)))hit=true;
        else if(dbl&&dist%d1===0&&dist/d1<=4){var ok=true;for(var st=1;st<dist/d1;st++)if(blocked(x+st*d1)){ok=false;break;}if(ok)hit=true;}}
      if(hit){var cost=0.3+(25-bi)/25*0.9+0.09*oh;if(cost>worst)worst=cost;}
    });
    tot+=worst*r.w;
  });
  return tot/36;
}
function evalP(p,s){
  var o=1-s,b=s*26,ob=o*26;
  if(p[b]===15)return 1;if(p[ob]===15)return 0;
  var pm=pip(p,s),po=pip(p,o);
  if(!contact(p)){
    var em=pm+1.2*(15-p[b]),eo=po+1.2*(15-p[ob]);
    var sd=Math.max(1.6,0.085*Math.min(em,eo)+1.2);
    return sig(((eo-4.1)-em)/sd*1.7);
  }
  var x=(po-pm)*0.012,i;
  var lm=backOf(p,s),lo=backOf(p,o);
  for(i=1;i<=24;i++){if(p[b+i]>=2)x+=PTV[i]*(i>lm?1:0.12);if(p[ob+i]>=2)x-=PTV[i]*0.9*(i>lo?1:0.12);if(p[b+i]>4)x-=(p[b+i]-4)*0.04;}
  x+=PRIME[bestPrime(p,s)]-PRIME[bestPrime(p,o)]*0.9;
  var hm=homePts(p,s),ho=homePts(p,o);
  x+=p[ob+25]*(0.22+0.12*hm)-p[b+25]*(0.22+0.12*ho);
  x+=(p[b]-p[ob])*0.05;
  /* checkers stuck behind a strong prime */
  var po_=bestPrime(p,o);if(po_>=4){var back=0;for(i=18;i<=25;i++)back+=p[b+i];x-=back*0.05*(po_-3);}
  x-=shotRisk(p,s)*1.15;
  x-=0.1; /* the other side rolls next */
  return sig(x);
}

  var sk=S.skill,p=onRollP(s,sk),race=!contact(S.pos);
  if(sk==='easy')return p>0.78&&R()<0.6;
  if(race)return p>=0.70;
  return p>=(sk==='hard'?0.68:0.71)&&p<0.92;
}
function botTakes(s){var sk=S.skill,p=evalP(S.pos,s);if(sk==='easy')return p>0.16||R()<0.3;return p>=(sk==='hard'?0.235:0.25);}

function waitFor(kind){return new Promise(function(res){S.pending={kind:kind,resolve:res};});}
function nextRoll(){var f=forceQ.shift();if(f)return f.slice(0,2);return [1+rnd(6),1+rnd(6)];}

function logLine(t,c){var li=document.createElement('li');if(c)li.className=c;li.innerHTML=t;var L0=$('log');L0.insertBefore(li,L0.firstChild);while(L0.children.length>160)L0.removeChild(L0.lastChild);}
async function rollFor(s){
  var g=S.gen,v=nextRoll();
  S.dice=v;S.left=v[0]===v[1]?[v[0],v[0],v[0],v[0]]:v.slice();S.openVals=null;
  S.roll={id:(S.roll?S.roll.id:0)+1,side:s,vals:v.slice()};S.phase='rolling';hostPush();updateControls();
  say(isMine(s)?'Rolling\u2026':esc(safeName(nameOf(s)))+' rolls\u2026');
  await throwFor(s,v);chk(g);
  if(v[0]===v[1]&&!isMine(s)&&R()<0.4)botTalk('bigroll',1);
}
function throwFor(s,v){var B=halfBox(s),from=s===ME?'bottom':'top';return Dice.throwDice([{i:0,val:v[0],B:B,from:from},{i:1,val:v[1],B:B,from:from}],opt.speed<0.7);}
async function openingRoll(){
  var g=S.gen;
  while(true){
    var f=forceQ.shift(),a=f?f[0]:1+rnd(6),b=f?f[1]:1+rnd(6); /* a = side 0's die, b = side 1's */
    S.phase='opening';S.openVals=[a,b];S.roll={id:(S.roll?S.roll.id:0)+1,side:-1,vals:[a,b]};hostPush();updateControls();
    say('Opening roll: one die each\u2026');
    var mv=ME===0?a:b,ov=ME===0?b:a;
    await Dice.throwDice([{i:0,val:mv,B:halfBox(ME),from:'bottom'},{i:1,val:ov,B:halfBox(1-ME),from:'top'}],opt.speed<0.7);chk(g);
    if(a!==b){S.turn=a>b?0:1;S.dice=[Math.max(a,b),Math.min(a,b)];S.left=S.dice.slice();
      say((isMine(S.turn)?'<b>You</b> go':'<b>'+esc(safeName(nameOf(S.turn)))+'</b> goes')+' first with '+S.dice[0]+'-'+S.dice[1]+'.');
      logLine('Opening roll: '+(S.turn===0?a+' beats '+b:b+' beats '+a)+' \u2013 '+esc(safeName(nameOf(S.turn)))+' starts','');
      await W(700);return;}
    say('Tie, '+a+' and '+b+'. Roll again.');await W(900);
  }
}
async function movesFor(s){
  var g=S.gen,c=S.ctl[s];
  S.phase='move';S.undo=[];S.turnMoves=[];LG=null;S.turnStart=S.pos.slice();select(-1);dimDice();updateControls();hostPush();
  var lg=legalNow(),rl=S.dice[0]+'-'+S.dice[1];
  if(lg.ml===0){
    say((isMine(s)?'You rolled ':esc(safeName(nameOf(s)))+' rolled ')+rl+'. <b>No legal moves.</b>');
    if(!isMine(s)&&S.pos[s*26+25])botTalk('dance',0.6);
    logLine(esc(safeName(nameOf(s)))+' '+rl+': no move','');
    await W(1300);return;
  }
  if(c==='bot'){
    say(esc(safeName(nameOf(s)))+' is thinking\u2026');await W(450);
    var seq=await botChoose(s,S.left.slice(),S.skill);chk(g);
    for(var i=0;i<seq.length;i++){var m=seq[i];var hit=await playMove(s,m);chk(g);S.left=rest(S.left,m.d);S.turnMoves.push({f:m.f,t:m.t,d:m.d,hit:hit});dimDice();hud();hostPush();
      if(hit)botTalk('hitYou',0.5);await W(170);}
  } else if(c==='me'){
    if(opt.forced){var fs=finals(S.pos,s,S.left);if(fs.length===1){say('Forced move \u2013 playing it for you.');await W(400);
      for(var j=0;j<fs[0].seq.length;j++){var m2=fs[0].seq[j];S.undo.push({pos:S.pos.slice(),left:S.left.slice(),moves:S.turnMoves.length});var h2=await playMove(s,m2);chk(g);S.left=rest(S.left,m2.d);S.turnMoves.push({f:m2.f,t:m2.t,d:m2.d,hit:h2});dimDice();LG=null;hud();hostPush();await W(120);}
      S.undo=[];updateControls();markUI();await W(250);logTurn(s,rl);return;}}
    say('Your move: '+leftText());markUI();updateControls();
    await waitFor('moves');chk(g);
  } else { /* remote guest */
    say('Waiting for '+esc(safeName(nameOf(s)))+' to move\u2026');
    var seqR=await waitFor('rmoves');chk(g);
    for(var k=0;k<seqR.length;k++){var m3=seqR[k];var h3=await playMove(s,m3,true);chk(g);S.left=rest(S.left,m3.d);S.turnMoves.push({f:m3.f,t:m3.t,d:m3.d,hit:h3});dimDice();hud();}
  }
  logTurn(s,rl);
}
function logTurn(s,rl){logLine('<b>'+esc(safeName(nameOf(s)))+'</b> '+rl+': '+(S.turnMoves.map(function(m){return notation(m,m.hit);}).join(' ')||'no move'),isMine(s)?'w':'');}
async function preroll(s){
  var g=S.gen,c=S.ctl[s];
  S.phase='preroll';S.dice=[];S.left=[];Dice.hide();S.openVals=null;updateControls();hostPush();

}
async function cubeOffer(s){
  var g=S.gen,t=1-s,c=S.ctl[t];
  S.offer=s;S.phase='cube';SFX.cube();layoutAll(true);updateControls();hostPush();
  logLine('<b>'+esc(safeName(nameOf(s)))+'</b> doubles to '+S.cube*2,'h');

  else if(c==='me'){if(S.ctl[s]==='bot')botTalk('double',0.9);showAsk(s);ans=await waitFor('cube');hideAsk();}
  else{say('Waiting for '+esc(safeName(nameOf(t)))+' to take or drop\u2026');ans=await waitFor('rcube');}

async function playGame(){
  var g=S.gen;
  try{
    await openingRoll();chk(g);
    await movesFor(S.turn);chk(g);
    while(true){
      if(winnerOf(S.pos)>=0){await finish(winnerOf(S.pos),resultMult(S.pos,winnerOf(S.pos)),'play');return;}
      S.turn=1-S.turn;
      var a=await preroll(S.turn);chk(g);
      if(a==='double'){var go=await cubeOffer(S.turn);chk(g);if(!go)return;}
      await rollFor(S.turn);chk(g);
      await movesFor(S.turn);chk(g);
    }
  }catch(e){if(e!==ABORT)console.error(e);}
}

  settle();hostPush();updateControls();

  if(iw){SFX.win();if(typeof suBanner==='function')suBanner(mult===3?'Backgammon!':mult===2?'Gammon!':'You Win!',S.stake?smoney(S.stake*pts):pts+' point'+(pts>1?'s':''));botTalk('lose',0.8);}

  S.escrow=0;hud();
}

function decideMode(){
  var L=LV.L,want='solo';
  if(L&&LV.enabled&&L.available()&&!AWAY&&L.status==='live'&&LV.roster.length>=2){if(!LV.settled)return;want=L.isHost()?'host':'guest';}
  if(want===LV.mode){renderLive();return;}
  if(LV.mode!=='solo')leaveLive(want==='solo');
  if(want!=='solo'){
    if(!idle()){ /* finish the computer game first */
      if(!LV.toldBusy){LV.toldBusy=true;toast((LV.others[0]?safeName(LV.others[0].name):'A player')+' sat down','Finish this game to play them');}
      clearTimeout(LV.decT);LV.decT=setTimeout(decideMode,1500);renderLive();return;}
    LV.toldBusy=false;enterLive(want);
  }

}

function startLiveGame(stake){
  stopGame();S.gid=(S.gid|0)+1+Math.floor(R()*1000);S.stake=stake;S.escrow=0;if(stake){bank-=stake;S.escrow=stake;}
  S.prop=null;S.phase='opening';resetBoard();hideSheet();S.names=[safeName(LV.L.me.name),LV.gName];
  logLine('Live game vs '+esc(LV.gName)+(stake?' \u2013 stake '+money(stake):' \u2013 friendly'),'h');updateControls();pushState();playGame();

function onAct(d,from){
  if(LV.mode!=='host'||!d||from!==LV.gId)return;
  if(typeof d.bank==='number'&&isFinite(d.bank))LV.oBank=d.bank;
  var p=S.pending;
  switch(d.kind){
    case 'accept':if(S.phase==='propose'&&S.prop){var st=S.prop.stake|0;if(st&&(bank<st*3||(LV.oBank!==null&&LV.oBank<st*3)))st=0;startLiveGame(st);}break;
    case 'decline':if(S.phase==='propose'){S.prop=null;S.phase='lobby';showSheet('lobby');toast(LV.gName+' declined','Pick another stake');pushState();}break;
    case 'roll':case 'double':if(d.gid===S.gid&&p&&p.kind==='rpreroll'){if(d.kind==='double'&&!canDouble(1)){pushState();break;}S.pending=null;p.resolve(d.kind);}else pushState(from);break;
    case 'take':case 'drop':if(d.gid===S.gid&&p&&p.kind==='rcube'){S.pending=null;p.resolve(d.kind);}break;
    case 'moves':if(d.gid===S.gid&&p&&p.kind==='rmoves'&&Array.isArray(d.moves)){var seq=validSeq(S.pos,1,S.left,d.moves);if(seq){S.pending=null;p.resolve(seq);}else{LV.L.send('reject',{gid:S.gid},from);pushState(from);}}break;
    case 'resign':if(d.gid===S.gid&&!idle()){stopGame();finish(0,1,'resign');}break;
  }
}
function validSeq(p,s,left,moves){
  var q=p.slice(),l=left.slice(),out=[],full=true;
  for(var i=0;i<moves.length;i++){var mv=moves[i];if(!Array.isArray(mv))return null;
    var lf=legalFirst(q,s,l,full),m=null;
    lf.moves.forEach(function(x){if(x.f===+mv[0]&&x.t===+mv[1]&&(!m||x.d===+mv[2]))m=x;});
    if(!m)return null;doMove(q,s,m);l=rest(l,m.d);out.push(m);full=false;}
  if(legalFirst(q,s,l,full).ml>0)return null;
  return out;
}
/* ---------- guest side: follow the host's state, animating rolls and moves ---------- */
function onState(st,from){
  if(LV.mode!=='guest'||!st||from!==LV.hostId||!Array.isArray(st.pos)||st.pos.length!==52)return;
  if(st.seq<=LV.lastSeq&&LV.lastFrom===from)return;LV.lastSeq=st.seq;LV.lastFrom=from;LV.lastSt=st;LV.stAt=Date.now();
  LV.q.push(st);if(LV.q.length>4)LV.q.splice(0,LV.q.length-1);pumpQ();
}
async function pumpQ(){if(LV.qBusy)return;LV.qBusy=true;try{while(LV.q.length&&LV.mode==='guest'){var st=LV.q.shift();await applyState(st,LV.q.length>1);}}catch(e){if(e!==ABORT)console.error(e);}LV.qBusy=false;}
function eqPos(a,b){for(var i=0;i<52;i++)if(a[i]!==+b[i])return false;return true;}

        else if(opt.forced){var fs=finals(S.pos,ME,S.left);if(fs.length===1){S.busy=true;say('Forced move \u2013 playing it for you.');
          try{for(var j=0;j<fs[0].seq.length;j++){var m2=fs[0].seq[j];var h2=await playMove(ME,m2);S.left=rest(S.left,m2.d);S.turnMoves.push({f:m2.f,t:m2.t,d:m2.d,hit:h2});LG=null;dimDice();}}finally{S.busy=false;}
          finishHumanTurn();}else say('Your move: '+leftText());}
        else say('Your move: '+leftText());}}
    if(S.phase==='preroll'){Dice.hide();if(S.turn===ME){if(prevPhase!=='preroll'){LV.sent=false;SFX.turn();if(opt.autoRoll&&!canDouble(ME)){LV.sent=true;setTimeout(function(){LV.L.send('act',{gid:S.gid,kind:'roll'},LV.hostId);},turbo?0:350);say('Your roll.');}else say('Your turn: <b>roll</b>'+(canDouble(ME)?' or <b>double</b>':'')+'.');}}

      say(iw?'<b>You win!</b>':'<b>'+esc(S.names[st.result.winner])+'</b> wins.');
      setTimeout(function(){if(LV.mode==='guest'&&S.phase==='over')showSheet('over');},turbo?0:1500);}}

  markUI();updateControls();
