function legal(G,s){

  }

  return out;
}

function endRound(G,w){
  var pp=G.seats.map(function(sd){return hpips(sd.hand);}),res={hand:G.hand,w:w,blocked:w<0,pips:pp,pts:[],mode:G.mode};
  if(G.mode==='classic'){
    if(w<0){var mn=Math.min.apply(null,pp),who=[];pp.forEach(function(p,i){if(p===mn)who.push(i);});w=who.length===1?who[0]:-1;res.w=w;}
    var pts=0;if(w>=0){pp.forEach(function(p,i){if(i!==w)pts+=p;});G.seats[w].score+=pts;}
    res.pts=G.seats.map(function(_,i){return i===w?pts:0;});

    if(top>=0){G.phase='over';G.win=top;}else G.phase='round';
  } else {

  }

}
/* apply one action for seat s. Returns an event or null if it is not allowed */
function apply(G,s,a){
  if(!G||G.phase!=='play'||G.turn!==s||!a)return null;
  var sd=G.seats[s],ev;
  if(a.t==='play'){
    var m=null;legal(G,s).forEach(function(x){if(x.k===a.k&&String(x.tg)===String(a.tg))m=x;});if(!m)return null;
    var hi=-1;sd.hand.forEach(function(t,i){if(tk(t)===a.k)hi=i;});if(hi<0)return null;
    var t=sd.hand.splice(hi,1)[0],dbl=t[0]===t[1];
    ev={kind:'play',seat:s,k:a.k,tg:m.tg,dbl:dbl?1:0};
    G.passes=0;G.lk[s]=[];
    if(G.mode==='classic'){
      if(m.tg==='C')G.c={a:t[0],b:t[1]};
      else{var e=cEnds(G),v=m.tg==='L'?e.L:e.R;(m.tg==='L'?G.L:G.R).push(t[0]===v?{a:t[0],b:t[1]}:{a:t[1],b:t[0]});}
      if(!sd.hand.length){ev.out=1;endRound(G,s);return ev;}
      adv(G);
    } else {
      var i=m.tg,v2=trEnd(G,i);G.tr[i].t.push(t[0]===v2?{a:t[0],b:t[1]}:{a:t[1],b:t[0]});
      if(G.tr[i].o===s&&sd.mk){sd.mk=false;ev.unmark=1;}
      if(G.tr[i].o>=0&&G.tr[i].o!==s)ev.onTrain=G.tr[i].o;
      var was=G.pend;
      if(!sd.hand.length){ev.out=1;endRound(G,s);return ev;}
      if(dbl){G.pend={tr:i,v:t[0],by:s};G.drew=false;G.dl=0;}
      else{if(was)ev.cover=was.by;G.pend=null;adv(G);}
    }
    return ev;
  }
  if(a.t==='draw'){
    if(!canDraw(G,s))return null;
    lack(G,s);
    var d=G.bone.pop();sd.hand.push(d);
    if(G.mode==='classic')G.drew++;else G.drew=true;
    return {kind:'draw',seat:s,k:tk(d),nd:G.mode==='classic'?G.drew:1};
  }
  if(a.t==='pass'){
    if(!mustPass(G,s))return null;
    lack(G,s);G.passes++;ev={kind:'pass',seat:s};
    if(G.mode==='mexican'&&!sd.mk){sd.mk=true;ev.mark=1;}
    if(!G.bone.length&&G.passes>=G.n){ev.blocked=1;endRound(G,-1);return ev;}
    adv(G);return ev;
  }
  return null;
}

function botPick(G,s,skill){
  var lg=legal(G,s);
  if(!lg.length)return canDraw(G,s)?{t:'draw'}:{t:'pass'};
  if(G.mode==='classic'&&!G.c)return {t:'play',k:lg[0].k,tg:'C'};
  if(skill==='easy'){var e=pick(lg);return {t:'play',k:e.k,tg:e.tg};}
  var h=G.seats[s].hand,best=null,bv=-1e9;
  lg.forEach(function(m){
    var t=pk(m.k),dbl=t[0]===t[1],v=t[0]+t[1]+(dbl?2.5:0)+R()*(skill==='hard'?0.6:1.6);
    if(G.mode==='classic'){
      if(skill==='hard'){
        var e2=cEnds(G),nv=(m.tg==='L'?e2.L:e2.R),out=t[0]===nv?t[1]:t[0],ends=[m.tg==='L'?out:e2.L,m.tg==='R'?out:e2.R];
        /* block: the next players are known to be out of these numbers */
        for(var j=1;j<G.n;j++){var o=(s+j)%G.n,w=j===1?2.2:1.1;ends.forEach(function(x){if(G.lk[o][x])v+=w*2;});}
        /* keep my options open: tiles left that still fit an end */
        var rest=h.filter(function(x){return tk(x)!==m.k;}),fit=0;rest.forEach(function(x){if(ends.indexOf(x[0])>=0||ends.indexOf(x[1])>=0)fit++;});
        v+=fit*1.4+(t[0]+t[1])*0.35;
      }
    } else {
      var own=G.tr[m.tg].o===s,others=G.tr[m.tg].o!==s;
      if(dbl){var cover=h.some(function(x){return tk(x)!==m.k&&(x[0]===t[0]||x[1]===t[0]);});v+=cover?(skill==='hard'?7:3):(skill==='hard'?-9:-2);}
      if(own&&G.seats[s].mk)v+=skill==='hard'?6:3;
      if(skill==='hard'){
        if(others&&G.tr[m.tg].o>=0)v+=2.5;           /* dump on someone else\u2019s open train */
        if(G.tr[m.tg].o<0)v+=1.2;
        if(G.pend)v+=1;
        /* don\u2019t use up the only tile that keeps my own train moving */
        if(!own){var myEnd=trEnd(G,s),keep=h.filter(function(x){return x[0]===myEnd||x[1]===myEnd;});if(keep.length===1&&tk(keep[0])===m.k)v-=4;}
      } else if(own)v+=0.8;
    }
    if(v>bv){bv=v;best=m;}
  });
  return {t:'play',k:best.k,tg:best.tg};
}

function runStep(){
  if(!ENG||!G||!G.seats)return;
  var key=evKey();if(key===stepKey)return;clearTimeout(stepT);stepKey=key;
  if(G.phase==='round'){
    if(DBG.auto||LV.mode==='host')stepT=later(DBG.auto?D(700):8000,function(){if(G&&G.phase==='round'&&ENG)nextRound();});
    return;
  }
  if(G.phase!=='play')return;
  var s=G.turn,sd=G.seats[s];
  if(G.mode==='classic'&&!G.c){stepT=later(D(sd.kind==='bot'?750:500),function(){if(G&&G.phase==='play'&&!G.c&&G.turn===s)doAct(s,{t:'play',k:G.lead,tg:'C'});});return;}
  if(sd.kind==='bot'){stepT=later(botDelay(),function(){if(G&&G.phase==='play'&&G.turn===s&&G.seats[s].kind==='bot')doAct(s,botPick(G,s,sd.skill||'normal'));});return;}
  /* a person: forced passes need no decision */
  if(mustPass(G,s)){stepT=later(D(950),function(){if(G&&G.turn===s&&mustPass(G,s))doAct(s,{t:'pass'});});}
  if(LV.mode!=='solo'&&!G.dl){G.dl=Date.now()+TURN_MS;publish();}
}
/* my own seat: autoplay (tests), draw for me */
function selfStep(){
  if(!G||!G.seats||G.phase!=='play'||!myTurn()||LV.sent)return;
  var key=evKey();if(key===selfKey)return;clearTimeout(selfT);selfKey=key;
  var s=mySeat();
  if(G.mode==='classic'&&!G.c)return;
  if(DBG.auto){selfT=later(D(320),function(){if(myTurn()&&evKey()===key)myAct(botPick(G,s,DBG.skill||'normal'));});return;}
  if(opt.autoDraw&&canDraw(G,s)){selfT=later(D(700),function(){if(myTurn()&&evKey()===key&&canDraw(G,s))myAct({t:'draw'});});}
}
function myAct(a){
  var s=mySeat();if(s<0||!G)return false;
  if(LV.mode==='guest'){
    var test=JSON.parse(JSON.stringify(G));if(!apply(test,s,a))return false;

    clearTimeout(LV.sentT);LV.sentT=setTimeout(function(){if(LV.sent){LV.sent=false;selfKey='';renderAll();status();selfStep();}},4000);
    renderAll();return true;
  }

}

  if(!G||G.phase!=='play')return false;

}

function newGame(seed){
  cancelSteps();
