    }

    if(!r.foul){

      else{var mine=r.assign||pre.groups[meS];r.cont=objs.some(function(n){return groupOf(n)===mine;});}
    }

  } else {

    if(!r.foul&&!pre.breakShot&&!pk.length&&!ev.railAfter)foul('No rail after contact');

    r.cont=!r.foul&&pk.length>0;
  }

}
