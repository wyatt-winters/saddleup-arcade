/* ball physics: scripted so it always lands in the pocket chosen in advance */
function launchBall(idx,quick){
  return new Promise(function(res){
    var b={phase:'track',idx:idx,quick:quick,t0:vt,done:res};
    b.Ta=quick?1.3:5.4;b.Td=quick?0.5:0.95;b.w0=-(quick?2.4:2.75)*TAU;b.wd=-(quick?1.25:1.05)*TAU;b.acc=(b.wd-b.w0)/b.Ta;b.a0=Math.random()*TAU;b.ang=b.a0;b.r=0.86;b.w=b.w0;
    WH.ball=b;WH.ot=quick?3.2:2.6;WH.otau=0.5;SFX.launch();rollStart();
  });
}
function ballStep(){
  var b=WH.ball;if(b.phase==='none')return false;
  var t=(vt-b.t0)/1000;
  if(b.phase==='track'){
    if(t<b.Ta){b.ang=b.a0+b.w0*t+0.5*b.acc*t*t;b.w=b.w0+b.acc*t;b.r=0.86+0.004*Math.sin(t*23)*(t/b.Ta);
      if(!roll&&canSnd())rollStart();var sp=Math.abs(b.w);rollSet(0.035+0.05*sp/17,350+sp*42,sp*1.4);return true;}
    b.phase='drop';b.aA=b.a0+b.w0*b.Ta+0.5*b.acc*b.Ta*b.Ta;b.t1=b.t0+b.Ta*1000;b.hit=false;noise(0.15,0.06,0,600,'lowpass');
  }
  if(b.phase==='drop'){
    var u=(vt-b.t1)/1000/b.Td;
    if(u<1){var hc=1/(1+Math.exp(-(u-0.42)/0.04));b.ang=b.aA+b.wd*b.Td*(u-0.25*u*u)+0.12*hc;b.w=b.wd*(1-0.5*u);
      b.r=0.86+(0.58*RK-0.86)*Math.pow(u,1.7)+0.04*Math.exp(-Math.pow((u-0.42)/0.08,2));
      if(!b.hit&&u>0.4){b.hit=true;SFX.clack(1);}rollSet(0.04*(1-u*0.5),500,Math.abs(b.w)*1.4);return true;}
    var wEnd=b.wd*0.5;b.ang=b.aA+b.wd*b.Td*0.75+0.12;
    b.rel0=b.ang-WH.phi;var target=b.idx*WH.step;b.v0=Math.max(2,Math.abs(wEnd-WH.omega));
    var dist=mod(b.rel0-target,TAU),minD=(b.quick?0.35:0.45)*b.v0;while(dist<minD)dist+=TAU;
    var T=b.quick?clamp(1.8*dist/b.v0,0.6,1.5):clamp(2.4*dist/b.v0,1.1,3.2);T=Math.min(T,3*dist/b.v0*0.97);
    b.D=dist;b.T=T;b.t2=vt;b.phase='pocket';b.lastK=null;b.hop=0;b.nh=b.quick?3:5;b.J=0.45*WH.step;WH.ot=0;WH.otau=7;
  }
  if(b.phase==='pocket'){
    var v=(vt-b.t2)/1000/b.T;
    if(v<1){var H1=v*v*v-2*v*v+v,H2=3*v*v-2*v*v*v,rel=b.rel0-(b.v0*b.T*H1+b.D*H2)+b.J*Math.sin(TAU*v*3.2)*v*(1-v)*(1-v)*4;
      b.ang=WH.phi+rel;b.w=-(b.v0*(1-v));b.r=RK*(0.515+0.065*Math.pow(1-v,1.5)*Math.abs(Math.cos(Math.PI*v*b.nh)));
      var kk=Math.round(rel/WH.step);if(b.lastK!==null&&kk!==b.lastK)SFX.tick(0.15+0.6*(1-v));b.lastK=kk;
      var hp=Math.floor(v*b.nh+0.5);if(hp!==b.hop){b.hop=hp;SFX.clack(0.25+0.6*(1-v));}
      rollSet(0.025*(1-v),900,6);return true;}
    b.phase='rest';b.rel=b.idx*WH.step;b.w=0;b.r=0.515*RK;rollStop();SFX.clack(0.35);WH.ot=0;WH.otau=4.5;var d=b.done;b.done=null;if(d)d();
  }
  if(b.phase==='rest'){b.ang=WH.phi+b.rel;b.r=0.515*RK;}
  return false;
}

/* where on the layout a chip at logical (lx,ly) goes: exact lines and corners like a real table */
function spotAt(lx,ly,touch){
  var f=touch?0.2:0.27,tx=clamp(f*B.ux,4,12)/B.ux,ty=clamp(f*B.uy,4,12)/B.uy,am=isAm();
  if(lx<-0.08||ly<-0.08||lx>14.08||ly>5.08)return null;
  if(ly>=4){if(lx<1||lx>=13)return null;return ['low','even','red','black','odd','high'][Math.floor((lx-1)/2)];}
  if(ly>3+ty){if(lx<1||lx>=13)return null;return 'd'+(Math.floor((lx-1)/4)+1);}
  if(lx>13+tx){if(ly>=3)return null;return 'c'+(3-clamp(Math.floor(ly),0,2));}
  if(lx<1-tx){if(ly>3)return null;if(am){if(Math.abs(ly-1.5)<ty)return ikey([0,37]);return ikey([ly<1.5?37:0]);}return ikey([0]);}
  var fx=lx-1,ex=Math.round(fx),dx=Math.abs(fx-ex),ey=Math.round(ly),dy=Math.abs(ly-ey);
  var nearV=dx<tx&&ex>=0&&ex<=11,nearH=dy<ty&&ey>=1&&ey<=3,col=clamp(Math.floor(fx),0,11),row=clamp(Math.floor(ly),0,2);
  if(nearV&&nearH){
    if(ex===0){if(ey===3)return ikey(am?[0,37,1,2,3]:[0,1,2,3]);return ey===1?ikey(am?[37,2,3]:[0,2,3]):ikey([0,1,2]);}
    if(ey===3){var sl=[];for(var c=ex-1;c<=ex;c++)for(var r=0;r<3;r++)sl.push(numAt(c,r));return ikey(sl);}
    return ikey([numAt(ex-1,ey-1),numAt(ex-1,ey),numAt(ex,ey-1),numAt(ex,ey)]);
  }
  if(nearV){
    if(ex===0){if(am){if(row===0)return ikey([37,3]);if(row===2)return ikey([0,1]);if(Math.abs(ly-1.5)<ty)return ikey([0,37,2]);return ikey([ly<1.5?37:0,2]);}return ikey([0,numAt(0,row)]);}
    return ikey([numAt(ex-1,row),numAt(ex,row)]);
  }
  if(nearH){if(ey===3)return ikey([numAt(col,0),numAt(col,1),numAt(col,2)]);return ikey([numAt(col,ey-1),numAt(col,ey)]);}
  return ikey([numAt(col,row)]);
}

function botPlan(b){
