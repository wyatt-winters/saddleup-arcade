     L.on('msg:<type>',fn(payload,fromId))
     L.on('status',fn(status))    'off' | 'connecting' | 'live' | 'error'

     L.send(type,payload)         broadcast to everyone at the table (not echoed to self)
     L.me -> {id,name}; L.isHost(); L.roster(); L.table (1,2,..); L.status
     L.setName(name)
