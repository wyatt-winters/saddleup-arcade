/* ---------- pots: pure distribution (used by the game and by tests) ---------- */
function buildPots(ps){
  var live=ps.filter(function(p){return !p.folded&&p.total>0;});
  var levels=[];live.forEach(function(p){if(levels.indexOf(p.total)<0)levels.push(p.total);});levels.sort(function(a,b){return a-b;});
  var pots=[],prev=0;
  levels.forEach(function(L){
    var amt=0;ps.forEach(function(p){amt+=Math.max(0,Math.min(p.total,L)-prev);});
    var el=live.filter(function(p){return p.total>=L;}).map(function(p){return p.i;});
    var last=pots[pots.length-1];
    if(last&&last.eligible.join()===el.join())last.amount+=amt;else pots.push({amount:amt,eligible:el});
    prev=L;
  });
  var extra=0;ps.forEach(function(p){extra+=Math.max(0,p.total-prev);});
  if(extra){if(pots.length)pots[pots.length-1].amount+=extra;else pots.push({amount:extra,eligible:[]});}
  return pots;
}
function distribute(ps,board,dealer){
  var N=ps.length,pay=ps.map(function(){return 0;}),pots=buildPots(ps),scores={};
  ps.forEach(function(p){if(!p.folded&&p.total>0&&p.hole&&p.hole.length===2&&board.length+2>=5)scores[p.i]=scoreCards(p.hole.concat(board));});
  var order=function(i){return (i-dealer-1+N*2)%N;};
  pots.forEach(function(pt){
    var el=pt.eligible,w;
    if(el.length===1)w=el.slice();
    else if(!el.length)w=[];
    else{var best=-1;el.forEach(function(i){if(scores[i]>best)best=scores[i];});w=el.filter(function(i){return scores[i]===best;});pt.score=best;}
    w.sort(function(a,b){return order(a)-order(b);});
    pt.winners=w;
    if(!w.length)return;
    var share=Math.floor(pt.amount/w.length),rem=pt.amount-share*w.length;
    w.forEach(function(i,k){pay[i]+=share+(k<rem?1:0);});
  });
  return {payouts:pay,pots:pots,scores:scores};
}

function labelFor(hole,board,folded,opp,hand,phase){
  var txt;
  if(board.length>=3)txt=handName(scoreCards(hole.concat(board)));
  else{var a=hole[0],b=hole[1],va=a.r===1?14:a.r,vb=b.r===1?14:b.r;
    if(va===vb)txt='Pocket '+RNP[va-2];else{var hi=Math.max(va,vb),lo=Math.min(va,vb);txt=RN[hi-2]+'-'+RN[lo-2]+(a.s===b.s?' suited':'');}}
  if(folded)return txt+' &middot; folded';
  var extra='';
  if(opt.odds&&phase!=='done'&&phase!=='showdown'){
    var key=hand+'|'+board.length+'|'+opp+'|'+hole.map(codeOf).join('');
    if(key!==oddsKey){oddsKey=key;oddsVal=turbo?null:equity(hole,board,opp,board.length===5?400:700);}
    if(oddsVal!==null)extra=' &middot; <b>'+Math.round(oddsVal*100)+'% win</b>';
  }
  return txt+extra;
}

function legal(i){
  var p=S.players[i],toCall=Math.max(0,S.currentBet-p.bet),maxTo=p.bet+p.chips;
  var others=S.players.filter(function(q,j){return j!==i&&canAct(q);}).length;
  var canRaise=!p.lock&&others>0&&p.chips>toCall&&maxTo>S.currentBet;
  var minTo=Math.min(S.currentBet+S.minRaise,maxTo);
  return {toCall:Math.min(toCall,p.chips),fullCall:toCall,canCheck:toCall===0,canRaise:canRaise,minTo:minTo,maxTo:maxTo};
}
function needs(p){
  if(!canAct(p))return false;
  if(p.acted&&p.bet>=S.currentBet)return false;
  if(countCanAct()<=1&&p.bet>=S.currentBet)return false;
  return true;
}
function nextToAct(from){for(var k=1;k<=5;k++){var j=(from+k)%5;if(needs(S.players[j]))return j;}return -1;}
function put(p,amt){amt=Math.min(amt,p.chips);p.chips-=amt;p.bet+=amt;p.total+=amt;if(p.chips===0)p.allIn=true;return amt;}
function applyAction(i,a){
  var p=S.players[i],L=legal(i),txt='',kind='',lg='',meta=a.meta||null,st=S.street,facing=L.fullCall>0;
  if(a.t==='fold'&&L.canCheck&&a.force!==true)a={t:'check'};
  if(a.t==='raise'&&!L.canRaise)a={t:L.canCheck?'check':'call'};
  if(a.t==='check'&&!L.canCheck)a={t:'call'};
  if(a.t==='fold'){p.folded=true;txt='Fold';kind='fold';lg='folds';}
  else if(a.t==='check'){txt='Check';kind='check';lg='checks';}
  else if(a.t==='call'){var amt=put(p,L.fullCall);txt=p.allIn?'All-in '+money(p.bet):'Call '+money(amt);kind=p.allIn?'allin':'call';lg=(p.allIn?'calls all-in ':'calls ')+money(amt);}
  else{
    var to=Math.round(a.to||L.minTo);to=clamp(to,L.minTo,L.maxTo);
    var prevCB=S.currentBet;put(p,to-p.bet);
    var inc=p.bet-prevCB,full=inc>=S.minRaise;
    if(full)S.minRaise=inc;
    if(p.bet>S.currentBet)S.currentBet=p.bet;
    S.players.forEach(function(q,j){if(j===i||!canAct(q))return;if(full){q.acted=false;q.lock=false;}else if(inc>0){if(q.acted)q.lock=true;q.acted=false;}});
    if(p.allIn){txt='All-in '+money(p.bet);kind='allin';lg='goes all-in for '+money(p.bet);}
    else if(prevCB===0){txt='Bet '+money(p.bet);kind='raise';lg='bets '+money(p.bet);}
    else{txt='Raise '+money(p.bet);kind='raise';lg='raises to '+money(p.bet);}
  }
  p.acted=true;p.last=txt;
  if(a.t==='raise'){S.lastAgg=i;if(st===0)S.pfAgg=i;p.aggStreets|=1<<st;}
  if(a.t==='check')p.checkedSt=st;
  if(isBot(p)){var B=BS[p.bot];
    if(st===0&&(a.t==='call'||a.t==='raise')&&!p.vpipH){p.vpipH=1;B.vpip++;}
    if(st===0&&a.t==='raise'&&!p.pfrH){p.pfrH=1;B.pfr++;}
    B[a.t==='fold'?'folds':a.t==='call'?'calls':a.t==='check'?'checks':'raises']++;
    if(st>0&&a.t==='raise'){B.postAgg++;if(meta&&meta.bluff)B.bluffs++;}
    if(st>0&&facing){B.faced++;if(a.t==='fold')B.foldToBet++;}
    if(a.t==='raise')p.bluffing=!!(meta&&meta.bluff);
    say(i,kind==='allin'?'allin':a.t,kind==='allin'?0.5:a.t==='fold'?0.1:0.13);
  } else {
    if(st===0&&(a.t==='call'||a.t==='raise')&&!p.vpipH){p.vpipH=1;HP.vpip++;}
    if(facing){HP.faced++;if(a.t==='fold')HP.foldTo++;}
    if(a.t==='raise')HP.agg++;else if(a.t==='call')HP.pas++;
  }
  setAct(i,txt,kind);
  log('<b>'+p.name+'</b> '+lg);
  if(kind==='fold')SFX.fold();else if(kind==='check')SFX.check();else SFX.chip();
  if((kind==='call'||kind==='raise'||kind==='allin')&&anim()&&els[i].bet.animate){
    var b=els[i].bet,s=LY.seat[i],t=LY.bet[i];b.classList.remove('z');
    b.animate([{transform:'translate(calc(-50% + '+(s.x-t.x)+'px),calc(-50% + '+(s.y-t.y)+'px)) scale(.6)',opacity:.3},{transform:'translate(-50%,-50%)',opacity:1}],{duration:260,easing:'cubic-bezier(.2,.8,.3,1)'});
  }
  return a;
}

/* preflop hand percentile from the Chen score over all 1326 starting hands (0 = best) */
var CHEN_PCT=(function(){var cnt={},tot=0,a,b;
  function cd(x){var v=(x>>2)+2;return {r:v===14?1:v,s:SUITS[x&3]};}
  for(a=0;a<52;a++)for(b=a+1;b<52;b++){var v=chen([cd(a),cd(b)]);cnt[v]=(cnt[v]||0)+1;tot++;}
  var pct={},keys=Object.keys(cnt).map(Number).sort(function(x,y){return y-x;}),acc=0;
  keys.forEach(function(k){pct[k]=(acc+cnt[k]/2)/tot;acc+=cnt[k];});return pct;})();
function handPct(h){var v=chen(h);return CHEN_PCT[v]!==undefined?CHEN_PCT[v]:1;}

function boardFeatures(hole,board){
  var f={fd:false,outs:0,overs:false,flush3:false,str4:false,paired:false,overNew:false,wet:false,scary:false,draw:false};
  if(board.length<3)return f;
  var v=function(c){return c.r===1?14:c.r;},sc={},bsc={},m=0,bm=0;
  hole.concat(board).forEach(function(c){sc[c.s]=(sc[c.s]||0)+1;m|=1<<(v(c)-2);});
  board.forEach(function(c){bsc[c.s]=(bsc[c.s]||0)+1;bm|=1<<(v(c)-2);});
  for(var s in sc){if(sc[s]===4&&(bsc[s]||0)<4&&board.length<5&&hole.some(function(c){return c.s===s;}))f.fd=true;}
  for(s in bsc){if(bsc[s]>=3)f.flush3=true;if(bsc[s]>=2&&board.length===3)f.wet=true;}
  function outs(mask){if(straightHi(mask)>=0)return 0;var n=0;for(var r=0;r<13;r++)if(!(mask&(1<<r))&&straightHi(mask|(1<<r))>=0)n++;return n;}
  if(board.length<5){var mo=outs(m),bo=outs(bm);f.outs=mo>bo?mo:0;}
  for(var lo=-1;lo<=8;lo++){var cntw=0;for(var k=0;k<5;k++){var r=lo+k,rr=r<0?12:r;if(bm&(1<<rr))cntw++;}if(cntw>=4)f.str4=true;if(cntw>=3)f.wet=true;}
  var seen={};board.forEach(function(c){if(seen[v(c)])f.paired=true;seen[v(c)]=1;});
  var mx=Math.max.apply(null,board.map(v));
  f.overs=board.length===3&&hole.every(function(c){return v(c)>mx;});
  if(board.length>3){var last=v(board[board.length-1]),prev=Math.max.apply(null,board.slice(0,-1).map(v));f.overNew=last>prev||last>=13;}
  if(f.flush3||f.str4)f.wet=true;
  f.scary=f.flush3||f.str4||f.paired||f.overNew;
  f.draw=f.fd||f.outs>=2||(f.overs&&board.length===3)||(f.outs===1&&f.fd===false&&Math.random()<0.4);
  return f;
}

function aiDecide(i){
  var p=S.players[i],L=legal(i),D=DIFF[opt.diff],P=p.pers,pot=potTotal(),bb=S.bb,R=Math.random();
  var opp=S.players.filter(function(q,j){return j!==i&&inHand(q);}).length;
  var humanIn=S.players.some(function(q){return !isBot(q)&&inHand(q);}),hp=hprof(),A=D.adapt,tilt=p.tilt>0;
  var adaptB=1,thin=0,light=0;
  if(humanIn&&hp.n>=6&&A>0){
    if(hp.foldTo>0.55)adaptB=Math.min(2.2,1+(hp.foldTo-0.55)*3.5*A);else if(hp.foldTo<0.35)adaptB=1-0.5*A;
    if(hp.vpip>0.45||hp.foldTo<0.32)thin=0.1*A;
    if(S.lastAgg>=0&&!isBot(S.players[S.lastAgg])&&hp.bluffy>0.3)light=0.08*A;
  }
  var ag=Math.min(0.97,P.aggr+(tilt?0.15:0));
  var out={t:'check'},meta={eq:null,bluff:false,kind:'',big:false,easy:false};
  function raiseTo(v){v=Math.round(v/S.sb)*S.sb;if(v>=L.maxTo*0.82)v=L.maxTo;return {t:'raise',to:clamp(v,L.minTo,L.maxTo)};}
  function betPot(f){return raiseTo(S.currentBet+Math.max(bb,(pot+L.fullCall)*f));}
  var commit=L.fullCall/Math.max(1,p.chips);
  if(S.street===0){
    var pct=clamp(handPct(p.hole)+gauss()*D.pnoise,0,1);
    var behind=0;if(i!==S.bbI){for(var k=1;k<5;k++){var j=(i+k)%5;if(canAct(S.players[j]))behind++;if(j===S.bbI)break;}}
    var posM=behind===0?1.35:behind===1?1.15:behind===2?0.95:0.8,tm=tilt?1.5:1;
    var vp=Math.min(0.9,P.vpip*posM*tm),pf=Math.min(0.7,P.pfr*posM*tm);
    var stackBB=(p.chips+p.bet)/bb,lvl=S.currentBet/bb;
    meta.easy=pct>vp*1.6&&!L.canCheck;
    if(stackBB<=10){
      if(pct<=Math.max(pf*1.7,0.06)+(10-stackBB)*0.025)out=L.canRaise?{t:'raise',to:L.maxTo}:{t:'call'};
      else if(L.canCheck)out={t:'check'};else if(L.fullCall<=bb/2&&pct<=vp)out={t:'call'};else out={t:'fold'};
      meta.big=out.t!=='fold'&&out.t!=='check';
    } else if(lvl<=1.001){
      var limpers=S.players.filter(function(q,j){return j!==S.bbI&&j!==S.sbI&&inHand(q)&&q.bet===S.currentBet;}).length;
      var open=pick(P.open)+limpers;
      if(pct<=pf&&L.canRaise)out=raiseTo(bb*open);
      else if(L.canCheck)out=(R<P.bluff*0.6*D.bluff&&L.canRaise)?(meta.bluff=true,raiseTo(bb*open)):{t:'check'};
      else if(pct<=vp)out={t:'call'};
      else if(behind<=1&&R<P.bluff*1.4*D.bluff*(tilt?1.6:1)&&L.canRaise){out=raiseTo(bb*open);meta.bluff=true;}
      else if(i===S.sbI&&pct<=vp*1.3)out={t:'call'};
      else out={t:'fold'};
    } else {
      var sq=Math.sqrt(lvl/3),def=vp*0.55/sq*(1+(0.4-P.fold)),tb=Math.max(0.025,pf*0.28/sq);
      if(i===S.bbI)def*=1.3;
      if(commit>0.4){def*=0.45;tb*=0.7;meta.big=true;}
      if(pct<=tb&&L.canRaise){var tg=S.currentBet*pick([2.8,3,3.3])+bb;out=tg>(p.chips+p.bet)*0.45?{t:'raise',to:L.maxTo}:raiseTo(tg);}
      else if(pct<=def)out={t:'call'};
      else if(L.fullCall<=bb/2&&pct<=vp)out={t:'call'};
      else if(L.canRaise&&commit<0.15&&R<P.bluff*0.35*D.bluff*(tilt?1.8:1)){out=raiseTo(S.currentBet*3);meta.bluff=true;}
      else out=L.canCheck?{t:'check'}:{t:'fold'};
      if(Math.abs(pct-def)<0.04)meta.big=true;
    }
  } else {
    var f=boardFeatures(p.hole,S.board);
    var eq=equity(p.hole,S.board,opp,turbo?Math.min(D.iters,160):D.iters)+gauss()*D.eqNoise;
    meta.eq=eq;
    var potBefore=Math.max(1,pot-L.fullCall),ratio=L.fullCall/potBefore;
    if(L.fullCall>0)eq-=Math.max(0,D.respect-light*4)*Math.min(0.12,ratio*0.1);
    var fair=1/(opp+1),str=(eq-fair)/(1-fair),ip=inPosition(i);
    var bm=D.bluff*adaptB*(ip?1.3:0.8)*(opp===1?1.3:opp===2?0.8:0.35)*(tilt?1.6:1);
    var cbetSpot=S.street===1&&S.pfAgg===i&&L.canCheck&&L.canRaise;if(cbetSpot)BS[p.bot].cbetChances++;
    if(L.canCheck){
      if(str>0.72){
        if(S.street<3&&!p.slow&&R<P.trap*(opp===1?1.2:0.7)){out={t:'check'};p.slow=true;meta.kind='slow';BS[p.bot].slowplays++;}
        else out=betPot(pick(P.sizes));
      }
      else if(p.slow&&str>0.55&&L.canRaise){out=betPot(pick([0.66,0.75,1]));p.slow=false;}
      else if(str>0.38-thin&&R<ag*0.85+thin&&L.canRaise)out=betPot(pick(P.sizes)*(thin&&str<0.45?0.75:1));
      else {
        var cbet=cbetSpot;
        var cprob=P.cbet*(f.wet?0.85:1.12)*(opp===1?1:opp===2?0.8:0.6)*(tilt?1.15:1)*Math.min(1.3,adaptB);
        var barrelled=S.street>=2&&(p.aggStreets&(1<<(S.street-1)));
        if(cbet&&R<cprob){out=betPot(pick([0.33,0.45,0.5,0.6]));meta.kind='cbet';}
        else if(barrelled&&(f.overNew||f.scary||f.draw)&&L.canRaise&&R<P.barrel*bm){
          out=(S.street===3&&R<P.overbet*4)?betPot(1.25+Math.random()*0.4):betPot(pick([0.6,0.7,0.8]));meta.kind='barrel';}
        else if(f.draw&&S.street<3&&L.canRaise&&R<(ag*0.3+P.bluff)*bm){out=betPot(pick([0.5,0.6,0.75]));meta.kind='semi';}
        else if(f.scary&&L.canRaise&&R<P.bluff*1.5*bm){out=betPot(pick([0.5,0.66,0.8]));meta.kind='bluff';}
        else if(ip&&opp===1&&L.canRaise&&S.lastAgg!==i&&R<P.bluff*0.8*bm){out=betPot(pick([0.4,0.5,0.6]));meta.kind='stab';}
        else if(S.street===3&&opp===1&&f.scary&&L.canRaise&&R<P.bigBluff*bm*1.5){out=betPot(1.3+Math.random()*0.5);meta.kind='bigbluff';}
        else out={t:'check'};
      }
    } else {
      var po=L.toCall/(pot+L.toCall);
      var margin=D.margin+(P.fold-0.4)*0.16*(1+Math.min(2,ratio))-light-(tilt?0.05:0)-(P.station||0);
      if(commit>0.5)margin+=0.1*P.fold;
      var need=po+margin;
      meta.big=commit>0.3||Math.abs(eq-need)<0.05;
      if(p.slow&&str>0.6&&L.canRaise){out=raiseTo(S.currentBet*pick([2.6,3,3.4])+pot*0.1);p.slow=false;meta.kind='trap';if(p.checkedSt===S.street)BS[p.bot].checkRaises++;}
      else if(str>0.66-ag*0.15&&L.canRaise){
        if(S.street<3&&R<P.trap*0.6){out={t:'call'};p.slow=true;BS[p.bot].slowplays++;}
        else{out=(str>0.85&&commit>0.3)?{t:'raise',to:L.maxTo}:raiseTo(S.currentBet*pick([2.3,2.6,3,3.2])+pot*0.15);if(p.checkedSt===S.street)BS[p.bot].checkRaises++;}
      }
      else if(eq>=need){
        if(f.draw&&L.canRaise&&S.street<3&&R<ag*0.22*bm){out=raiseTo(S.currentBet*2.7+pot*0.2);meta.kind='semi';}
        else out={t:'call'};
      }
      else if(L.canRaise&&p.checkedSt===S.street&&opp===1&&(f.scary||f.draw)&&commit<0.35&&R<P.checkRaise*1.6*bm){out=raiseTo(S.currentBet*2.8+pot*0.15);meta.kind='crbluff';BS[p.bot].checkRaises++;}
      else if(L.canRaise&&opp===1&&S.street===3&&commit<0.35&&R<P.bluff*0.25*bm){out=raiseTo(S.currentBet*2.6+pot*0.2);meta.kind='bluff';}
      else out={t:'fold'};
    }
    if(out.t==='raise'){
      if(eq<0.3){meta.bluff=true;if(meta.kind==='semi'||f.draw)BS[p.bot].semi++;}
      if(cbetSpot)BS[p.bot].cbets++;
    }
    meta.easy=(out.t==='check'&&str<0.05)||(out.t==='call'&&eq>need+0.25);
  }
  if(out.t==='fold'&&L.canCheck)out={t:'check'};
  if(out.t==='raise'&&!L.canRaise)out={t:L.canCheck?'check':'call'};
  if(out.t==='raise'&&out.to>=L.maxTo)meta.big=true;
  out.meta=meta;
  return out;
}
